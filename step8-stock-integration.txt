const StockReservationService = require(
    "../../../application/StockReservationService"
);

function handleError(res, error) {
    console.error("Stock Reservation Error:", error.message);

    const status = Number.isInteger(error.status)
        ? error.status
        : 500;

    return res.status(status).json({
        success: false,
        message: status === 500
            ? "Internal server error"
            : error.message
    });
}



async function reserve(req, res) {
  try {
    const {
      reservationId,
      orderId,
      productId,
      quantity
    } = req.body || {};

    const result = await StockReservationService.reserveStock({
      reservationId,
      orderId,
      productId,
      quantity
    });

    return res.status(result.alreadyExists ? 200 : 201).json({
      success: true,
      message: result.alreadyExists
        ? "Reservation already exists"
        : "Stock reserved successfully",
      data: result.reservation
    });
  } catch (error) {
    return handleError(res, error);
  }
}



async function commit(req, res) {
    try {
        const { reservationId } = req.body;

        const reservation =
            await StockReservationService.commitReservation(
                reservationId
            );

        return res.status(200).json({
            success: true,
            message: "Reservation committed",
            data: reservation
        });
    } catch (error) {
        return handleError(res, error);
    }
}

async function release(req, res) {
    try {
        const { reservationId } = req.body;

        const reservation =
            await StockReservationService.releaseReservation(
                reservationId
            );

        return res.status(200).json({
            success: true,
            message: "Reservation released",
            data: reservation
        });
    } catch (error) {
        return handleError(res, error);
    }
}

async function getById(req, res) {
    try {
        const reservation =
            await StockReservationService.getReservation(
                req.params.id
            );

        return res.status(200).json({
            success: true,
            data: reservation
        });
    } catch (error) {
        return handleError(res, error);
    }
}

module.exports = {
    reserve,
    commit,
    release,
    getById
};

const express = require("express");
const crypto = require("node:crypto");

const controller = require(
    "../controllers/StockReservationController"
);

const router = express.Router();

router.use((req, res, next) => {
    const expected = process.env.INTERNAL_SERVICE_KEY;
    const provided = req.get("x-internal-service-key");

    if (!expected || !provided) {
        return res.status(401).json({
            success: false,
            message: "Unauthorized internal request"
        });
    }

    const a = Buffer.from(expected);
    const b = Buffer.from(provided);

    if (
        a.length !== b.length ||
        !crypto.timingSafeEqual(a, b)
    ) {
        return res.status(401).json({
            success: false,
            message: "Unauthorized internal request"
        });
    }

    next();
});

router.post("/reserve", controller.reserve);
router.post("/commit", controller.commit);
router.post("/release", controller.release);
router.get("/reservations/:id", controller.getById);

module.exports = router;

===== services\product-service\src\application\StockReservationService.js =====

const mongoose = require("mongoose");
const Product = require("../infrastructure/database/models/ProductModel");
const StockReservation = require("../infrastructure/database/models/StockReservationModel");

function fail(message, status = 400) {
  const error = new Error(message);
  error.status = status;
  throw error;
}

function validReservationId(value) {
  if (typeof value !== "string" ||
      value.length < 1 ||
      value.length > 100 ||
      value.trim() !== value) {
    fail("Invalid reservationId");
  }
}

function sameReservation(record, data) {
  return record.orderId === data.orderId &&
    String(record.productId) === String(data.productId) &&
    record.quantity === data.quantity;
}

async function reserveStock({
  reservationId,
  orderId,
  productId,
  quantity
}) {
  validReservationId(reservationId);

  if (typeof orderId !== "string" ||
      !orderId.trim() ||
      orderId.length > 100) {
    fail("Invalid orderId");
  }

  if (!mongoose.isValidObjectId(productId)) {
    fail("Invalid productId");
  }

  if (!Number.isSafeInteger(quantity) || quantity <= 0) {
    fail("Invalid quantity");
  }

  const input = {
    reservationId,
    orderId,
    productId,
    quantity
  };

  const existing = await StockReservation.findOne({
    reservationId
  });

  if (existing) {
    if (!sameReservation(existing, input)) {
      fail("Reservation ID conflict", 409);
    }

    return {
      reservation: existing,
      alreadyExists: true
    };
  }

  try {
    const reservation = await mongoose.connection.transaction(
      async (session) => {
        const product = await Product.findOneAndUpdate(
        {
            _id: productId,
            isActive: true,
            stock: { $gte: quantity }
        },
        {
            $inc: { stock: -quantity }
        },
        {
            returnDocument: "after",
            session
        }
        );

        if (!product) {
          fail("Insufficient stock or inactive product", 409);
        }

        const records = await StockReservation.create(
          [{
            reservationId,
            orderId,
            productId,
            quantity,
            status: "reserved",
            reservedAt: new Date()
          }],
          { session }
        );

        return records[0];
      }
    );

    return {
      reservation,
      alreadyExists: false
    };
  } catch (error) {
    if (error.code === 11000) {
      const found = await StockReservation.findOne({
        reservationId
      });

      if (found && sameReservation(found, input)) {
        return {
          reservation: found,
          alreadyExists: true
        };
      }

      fail("Reservation ID conflict", 409);
    }

    throw error;
  }
}

async function commitReservation(reservationId) {
  validReservationId(reservationId);

  return mongoose.connection.transaction(async (session) => {
    const record = await StockReservation.findOne({
      reservationId
    }).session(session);

    if (!record) {
      fail("Reservation not found", 404);
    }

    if (record.status === "committed") {
      return record;
    }

    if (record.status !== "reserved") {
      fail("Reservation cannot be committed", 409);
    }

    record.status = "committed";
    record.committedAt = new Date();

    await record.save({ session });

    return record;
  });
}

async function releaseReservation(reservationId) {
  validReservationId(reservationId);

  return mongoose.connection.transaction(async (session) => {
    const record = await StockReservation.findOne({
      reservationId
    }).session(session);

    if (!record) {
      fail("Reservation not found", 404);
    }

    if (record.status === "released") {
      return record;
    }

    if (record.status !== "reserved") {
      fail("Reservation cannot be released", 409);
    }

    const result = await Product.updateOne(
      { _id: record.productId },
      { $inc: { stock: record.quantity } },
      { session }
    );

    if (result.matchedCount !== 1) {
      fail("Product not found during stock release", 409);
    }

    record.status = "released";
    record.releasedAt = new Date();

    await record.save({ session });

    return record;
  });
}

async function getReservation(reservationId) {
  validReservationId(reservationId);

  const record = await StockReservation.findOne({
    reservationId
  });

  if (!record) {
    fail("Reservation not found", 404);
  }

  return record;
}

module.exports = {
  reserveStock,
  commitReservation,
  releaseReservation,
  getReservation
};
